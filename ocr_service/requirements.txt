# NSRP handwriting scanner — Python dependencies.
#
# Only what service.py and pipeline.py actually import. Everything else in the
# virtualenv arrives as a dependency of these.
#
# Do NOT regenerate this file with `pip freeze`. Two of the lines below differ
# deliberately from what is installed on the Windows development machine, and a
# freeze would undo both:

# 1. torch must come from PyTorch's own index, not PyPI.
#    Plain `pip install torch` on Linux pulls the CUDA build — about 2.5 GB of
#    GPU driver for a server that has no GPU, and enough to fill a small disk.
--extra-index-url https://download.pytorch.org/whl/cpu

# 2. opencv-python-headless, never plain opencv-python.
#    The desktop build links against GUI libraries a headless server does not
#    have, and the import dies with:
#        ImportError: libGL.so.1: cannot open shared object file
#    Same API, same code, no window system.

flask==3.1.3
numpy==2.5.2
opencv-python-headless==5.0.0.93
pillow==12.3.0
torch==2.13.0
